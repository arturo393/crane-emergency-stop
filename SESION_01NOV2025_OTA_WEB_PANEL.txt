ESTRUCTURA DE CÓDIGO
================================================================================

OTA MANAGER:
-----------
class OtaManager {
    // Actualización
    OtaResult update_from_url(url, sha256, callbacks, timeout);
    OtaResult update_from_buffer(data, size, callbacks);
    
    // Validación
    esp_err_t mark_as_valid();
    bool is_pending_validation();
    
    // Rollback
    esp_err_t rollback();
    
    // Info
    esp_err_t get_running_partition_info(partition, version);
    OtaState get_state();
    
    // Control
    void cancel_update();
};

WEB SERVER:
----------
struct GatewayStatus {
    // CAN
    bool can_online;
    uint32_t can_rx_count, can_tx_count, can_errors;
    
    // CiA402
    uint8_t cia402_state;
    uint16_t status_word, control_word;
    
    // Network
    bool eth_connected;
    char ip_address[16];
    uint16_t tcp_port;
    uint8_t clients_connected;
    
    // System
    uint32_t uptime_seconds, free_heap;
    uint8_t cpu_usage;
    
    // SD
    bool sd_mounted;
    uint32_t sd_total_mb, sd_used_mb;
    uint8_t log_files_count;
    
    // OTA
    char firmware_version[32];
    char ota_partition[17];
    bool ota_pending;
};

class WebServer {
    esp_err_t init(port = 80);
    esp_err_t stop();
    void update_status(GatewayStatus);
    void set_status_callback(callback);
};
